{{- /* Site override of the theme image hook. The theme only set data-src (lazysizes), so without JavaScript – and for image search – posts had no images at all. Use a real src with native lazy loading, add width/height for page-bundle images (stops layout shift), and fall back to the post title when the Markdown has no alt text. */ -}} {{- $src := .Destination -}} {{- $res := false -}} {{- if not (urls.Parse $src).IsAbs -}} {{- $res = or (.Page.Resources.GetMatch $src) (.Page.Resources.GetMatch (printf "**%s" (path.Base $src))) -}} {{- end -}} {{- $alt := .PlainText | default .Page.Title -}}
